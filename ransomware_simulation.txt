# Simulação educativa de criptografia de arquivos
# NÃO é ransomware real

from cryptography.fernet import Fernet

# Gerar chave
key = Fernet.generate_key()
cipher = Fernet(key)

# Arquivo de teste
file_name = "teste.txt"

# Criar arquivo de exemplo
with open(file_name, "w") as f:
    f.write("Este é um arquivo de teste.")

# Ler conteúdo
with open(file_name, "rb") as f:
    data = f.read()

# Criptografar
encrypted_data = cipher.encrypt(data)

# Salvar criptografado
with open(file_name, "wb") as f:
    f.write(encrypted_data)

print("Arquivo criptografado (simulação)")

# Descriptografar (para mostrar reversão)
decrypted_data = cipher.decrypt(encrypted_data)

with open(file_name, "wb") as f:
    f.write(decrypted_data)

print("Arquivo restaurado (simulação)")
